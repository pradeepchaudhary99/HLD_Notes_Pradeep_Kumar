▸ view backend hold logic (Java / Lettuce)
// Atomic hold — succeeds only if seat isn't already locked
boolean holdSeat(String seatId, String userId) {
    long expiresAt = System.currentTimeMillis() + HOLD_TTL_MS;
    Long added = redis.zadd(
        "seat_holds:showtime:42",
        ZAddArgs.Builder.nx(),   // NX = only if absent
        expiresAt,                    // score = expiry timestamp
        seatId                        // member
    );
    return added != null && added == 1L;   // 1 = lock acquired, 0 = already held
}

// Sweeper: cheaply finds only the holds that expired (sorted by score!)
void releaseExpiredHolds() {
    long now = System.currentTimeMillis();
    Set<String> expired = redis.zrangebyscore("seat_holds:showtime:42", 0, now);
    if (!expired.isEmpty()) {
        redis.zremrangebyscore("seat_holds:showtime:42", 0, now);
    }
}

// Confirm: drop the hold, persist the booking
void confirmBooking(String seatId) {
    redis.zrem("seat_holds:showtime:42", seatId);
    db.markSeatBooked(seatId);
}

Redis: ZSET seat_holds

member = seatId  ·  score = hold-expiry timestamp  ·  sorted ascending by score
Sorted set is empty — no active holds
Because entries are ordered by expiry score, a background sweeper never scans the whole set — it just asks ZRANGEBYSCORE seat_holds 0 now, which is O(log N + M). The NX flag on ZADD makes "check-and-lock" atomic, so two users clicking the same seat in the same millisecond can't both win — Redis's single-threaded execution guarantees only one ZADD returns 1.
